"""Essai ciblé de T-0171 : des clones locaux d'une origine pendant que d'autres
clones y poussent, comme l'auto-test ; compte les clones qui échouent.
Usage : python3 course.py <mode: local|no-local|no-hardlinks> <rondes>"""
import os, subprocess, sys, tempfile, threading
from pathlib import Path
from concurrent.futures import ThreadPoolExecutor

mode, rondes = sys.argv[1], int(sys.argv[2])
env = {k: v for k, v in os.environ.items() if not k.startswith("GIT_")}
env.update({"GIT_AUTHOR_NAME": "a", "GIT_AUTHOR_EMAIL": "a@a", "GIT_COMMITTER_NAME": "a",
            "GIT_COMMITTER_EMAIL": "a@a", "GIT_CONFIG_NOSYSTEM": "1",
            "GIT_CONFIG_COUNT": "3", "GIT_CONFIG_KEY_0": "gc.auto", "GIT_CONFIG_VALUE_0": "0",
            "GIT_CONFIG_KEY_1": "maintenance.auto", "GIT_CONFIG_VALUE_1": "false",
            "GIT_CONFIG_KEY_2": "receive.autogc", "GIT_CONFIG_VALUE_2": "false"})
d = Path(tempfile.mkdtemp(prefix="course-"))
env["HOME"] = str(d)
def git(cwd, *a):
    return subprocess.run(["git", *a], cwd=cwd, env=env, capture_output=True, text=True)
o = d / "origine"; o.mkdir()
git(o, "init", "-q", "-b", "main")
for i in range(200):
    (o / f"f{i}.txt").write_text(f"{i}\n" * 50)
git(o, "add", "-A"); git(o, "commit", "-q", "-m", "init")
opts = {"local": [], "no-local": ["--no-local"], "no-hardlinks": ["--no-hardlinks"]}[mode]
n = 0; verrou = threading.Lock(); echecs = []
def nouveau():
    global n
    with verrou:
        n += 1; return d / f"cas-{n}"
def cloner():
    c = nouveau()
    r = git(d, "clone", "-q", *opts, str(o), str(c))
    if r.returncode:
        echecs.append(r.stderr.strip().splitlines()[-1][:160])
        return None
    return c
def pousser(k):
    c = cloner()
    if c is None:
        return
    for j in range(3):
        for q in range(150):
            (c / f"n{k}-{j}-{q}.txt").write_text(os.urandom(2000).hex())
        git(c, "add", "-A"); git(c, "commit", "-q", "-m", f"{k}-{j}")
        git(c, "push", "-q", "origin", f"HEAD:refs/heads/b{k}-{j}")
taches = []
for k in range(rondes):
    taches += [("p", k), ("c", k), ("p", k), ("c", k), ("c", k), ("c", k)]
with ThreadPoolExecutor(max_workers=4) as ex:
    list(ex.map(lambda t: pousser(t[1]) if t[0] == "p" else cloner(), taches))
clones = n
print(f"{mode}: {len(echecs)} clone(s) en échec sur {clones}")
for e in sorted(set(echecs))[:5]:
    print("  ", e)
subprocess.run(["rm", "-rf", str(d)])
